func TestAuditWitnessFileHandlersAcceptForeignObject(t *testing.T) {
    const object = "drive/user-999/private.txt"
    const payload = "synthetic foreign document"
    deleted := ""
    storage := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        if r.Method == http.MethodDelete { deleted = r.URL.Path; w.WriteHeader(204); return }
        if strings.HasSuffix(r.URL.Path, object) {
            w.Header().Set("Content-Type", "text/plain")
            w.Header().Set("Content-Length", fmt.Sprint(len(payload)))
            w.Header().Set("Last-Modified", time.Now().UTC().Format(http.TimeFormat))
            w.Header().Set("ETag", `"synthetic-etag"`)
            if r.Method == http.MethodGet { _, _ = w.Write([]byte(payload)) }
            return
        }
        w.WriteHeader(200)
    }))
    defer storage.Close()
    fs, err := service.NewFileService(config.MinIOConfig{Endpoint: strings.TrimPrefix(storage.URL, "http://"), AccessKey: "synthetic-access", SecretKey: "synthetic-secret", Bucket: "audit", Region: "us-east-1", PublicBaseURL: storage.URL+"/audit"}, "http://workspace.test", nil, zap.NewNop())
    if err != nil { t.Fatal(err) }
    h := NewFileHandler(fs, zap.NewNop())
    router := gin.New()
    router.DELETE("/api/v1/files/public", h.DeletePublicFile)
    router.GET("/api/v1/files/download", h.DownloadFile)
    router.GET("/api/v1/files/url", func(c *gin.Context) { c.Set("user_id", int64(10)); h.GetFileURL(c) })
    router.DELETE("/api/v1/files", func(c *gin.Context) { c.Set("user_id", int64(10)); h.DeleteFile(c) })
    query := "?"+url.Values{"object_name": {object}}.Encode()
    response := httptest.NewRecorder()
    router.ServeHTTP(response, httptest.NewRequest("GET", "/api/v1/files/url"+query, nil))
    if response.Code != 200 { t.Fatalf("url status=%d body=%s", response.Code, response.Body.String()) }
    var result map[string]any
    if err := json.Unmarshal(response.Body.Bytes(), &result); err != nil { t.Fatal(err) }
    if data, ok := result["data"].(map[string]any); ok { result = data }
    signed, ok := result["url"].(string)
    if !ok { t.Fatalf("URL missing: %s", response.Body.String()) }
    link, err := url.Parse(signed)
    if err != nil { t.Fatal(err) }
    response = httptest.NewRecorder()
    router.ServeHTTP(response, httptest.NewRequest("GET", link.RequestURI(), nil))
    if response.Code != 200 || response.Body.String() != payload { t.Fatalf("download status=%d body=%s", response.Code, response.Body.String()) }
    t.Log("CONFIRMED: user 10 obtained a signature for a foreign object; anonymous signed download returned the synthetic content")
    for _, path := range []string{"/api/v1/files/public", "/api/v1/files"} {
        deleted = ""
        response = httptest.NewRecorder()
        router.ServeHTTP(response, httptest.NewRequest("DELETE", path+query, nil))
        if response.Code != 200 || deleted != "/audit/"+object { t.Fatalf("delete %s status=%d storage_path=%s", path, response.Code, deleted) }
        t.Logf("CONFIRMED: %s deleted the foreign synthetic storage object", path)
    }
}

func TestAuditWitnessNegativeConsumptionIncreasesStock(t *testing.T) {
